FIX 2 — Add updateLockedSection() function:

Add this new function before bindEvents():

function updateLockedSection() {
  const lockedSection = document.querySelector('.locked-section');
  if (!lockedSection) return;

  if (authState.signedIn) {
    // User is signed in — transform locked section into full audit CTA
    lockedSection.innerHTML = `
      <div class="locked-header" style="background:rgba(34,197,94,0.08);
           border-bottom:1px solid rgba(34,197,94,0.15);">
        <div class="locked-title" style="color:#4ade80;">
          ✅ Signed in as ${authState.email}
        </div>
        <div class="locked-subtitle">
          Run your full audit to see E-E-A-T, Google Knowledge Graph, 
          Reddit presence, 100 content topics and live citation check.
        </div>
      </div>
      <div style="padding:14px;">
        <a id="signed-in-full-audit" href="#"
           style="display:block;background:#6366f1;color:#fff;
                  text-align:center;padding:11px;border-radius:8px;
                  text-decoration:none;font-weight:700;font-size:13px;
                  margin-bottom:8px;">
          Run Full Audit — See All Signals →
        </a>
        <div style="font-size:11px;color:#475569;text-align:center;
                    line-height:1.5;">
          Opens Citation Rocket with this page pre-loaded.<br>
          Your ${authState.plan || 'free'} plan features will be available.
        </div>
      </div>
    `;

    // Wire up the full audit button
    const btn = document.getElementById('signed-in-full-audit');
    if (btn && currentAudit) {
      btn.href = `${CR_APP}/dashboard?url=${encodeURIComponent(currentAudit.pageUrl || 'https://' + currentAudit.domain)}`;
      btn.setAttribute('target', '_blank');
    } else if (btn) {
      btn.href = CR_APP + '/dashboard';
      btn.setAttribute('target', '_blank');
    }
  }
  // If not signed in, leave existing locked section as-is
}

FIX 3 — Add showRescanPrompt() function:

Add this function right after updateLockedSection():

function showRescanPrompt() {
  // After sign-in with existing scan, show a subtle 
  // "rescan for updated results" notice above the score
  const hero = document.querySelector('.score-hero');
  if (!hero) return;

  // Remove existing prompt if any
  const existing = document.getElementById('rescan-notice');
  if (existing) existing.remove();

  const notice = document.createElement('div');
  notice.id = 'rescan-notice';
  notice.style.cssText = `
    background: rgba(99,102,241,0.1);
    border: 1px solid rgba(99,102,241,0.2);
    border-radius: 8px;
    padding: 10px 14px;
    margin: 12px 16px 0;
    display: flex;
    align-items: center;
    justify-content: space-between;
    gap: 10px;
    font-size: 12px;
    color: #818cf8;
  `;
  notice.innerHTML = `
    <span>You're now signed in. Run a new scan to see your plan features.</span>
    <button onclick="document.getElementById('scan-btn').click();
                     this.parentElement.remove();"
            style="background:#6366f1;color:#fff;border:none;
                   padding:5px 12px;border-radius:6px;font-size:11px;
                   font-weight:700;cursor:pointer;white-space:nowrap;">
      Re-scan
    </button>
  `;

  // Insert before the score hero
  hero.parentNode.insertBefore(notice, hero);
}
