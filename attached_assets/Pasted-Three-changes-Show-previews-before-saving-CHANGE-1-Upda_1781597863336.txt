// POST /api/extension/analyze
// Called by the extension after sign-in to get full 
// E-E-A-T, off-site, and blueprint scores using page 
// content sent from the browser. No re-scraping needed.
app.post('/api/extension/analyze', requireAuth, async (req: AuthenticatedRequest, res) => {
  try {
    const { url, html, text, domain } = req.body;
    
    if (!url || !html) {
      return res.status(400).json({ 
        error: 'url and html are required' 
      });
    }

    const { runBlueprintAudit } = await import('./blueprint-orchestrator');
    
    // Extract company name from domain
    const domainName = domain || new URL(url).hostname.replace(/^www\./, '');
    const companyName = domainName.split('.')[0].charAt(0).toUpperCase() + 
                        domainName.split('.')[0].slice(1);

    const blueprint = await runBlueprintAudit({
      websiteUrl: url,
      html: html || '',
      text: text || '',
      sitemapUrls: [],
      brandName: companyName,
      companyName,
      industry: req.body.industry || '',
      primaryQuery: '',
    });

    return res.json({ 
      success: true, 
      blueprint,
      companyName,
    });

  } catch (err: any) {
    console.error('Extension analyze error:', err);
    return res.status(500).json({ 
      error: 'Analysis failed: ' + err.message 
    });
  }
});

async function fetchSignedInData() {
  if (!authState.signedIn || !authState.authToken || !currentAudit) return;
  
  // Show loading state in locked section
  const lockedSection = document.querySelector('.locked-section');
  if (!lockedSection) return;
  
  lockedSection.innerHTML = `
    <div class="locked-header">
      <div class="locked-title" style="color:#818cf8;">
        Loading your full analysis...
      </div>
      <div style="display:flex;align-items:center;gap:8px;
                  padding:12px 0 4px;">
        <div style="width:20px;height:20px;border:2px solid 
                    rgba(99,102,241,0.2);border-top-color:#6366f1;
                    border-radius:50%;animation:spin 0.8s linear infinite;">
        </div>
        <span style="font-size:12px;color:#64748b;">
          Running E-E-A-T, Knowledge Graph and Reddit checks...
        </span>
      </div>
    </div>
  `;

  try {
    const response = await fetch(`${CR_APP}/api/extension/analyze`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${authState.authToken}`,
      },
      body: JSON.stringify({
        url: currentAudit.pageUrl,
        html: currentAudit._html || '',
        text: currentAudit._text || '',
        domain: currentAudit.domain,
      }),
    });

    if (!response.ok) {
      throw new Error('Server returned ' + response.status);
    }

    const data = await response.json();
    if (data.success && data.blueprint) {
      renderSignedInResults(data.blueprint, lockedSection);
    } else {
      throw new Error('No blueprint data in response');
    }

  } catch (err: any) {
    lockedSection.innerHTML = `
      <div class="locked-header">
        <div class="locked-title" style="color:#f87171;">
          ⚠️ Could not load full analysis
        </div>
        <div class="locked-subtitle">${err.message}</div>
      </div>
    `;
  }
}

async function loadAuthState() {
  return new Promise(resolve => {
    chrome.storage.local.get(
      ['authToken', 'userEmail', 'userPlan'], 
      (data) => {
        if (data.authToken) {
          authState = {
            signedIn: true,
            email: data.userEmail || '',
            plan: data.userPlan || 'free',
            authToken: data.authToken,
          };
        }
        resolve(undefined);
    });
  });
}
