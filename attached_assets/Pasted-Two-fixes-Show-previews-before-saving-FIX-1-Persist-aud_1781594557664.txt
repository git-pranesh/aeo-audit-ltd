In getCurrentTab() — after getting the tab, check for a 
cached audit for this domain and restore it:

async function getCurrentTab() {
  try {
    const [tab] = await chrome.tabs.query({ 
      active: true, currentWindow: true 
    });
    currentTab = tab || null;
    
    // Restore cached audit for this domain if it exists
    if (currentTab?.url) {
      try {
        const domain = new URL(currentTab.url).hostname.replace('www.', '');
        const cached = await chrome.storage.local.get(`audit_${domain}`);
        if (cached[`audit_${domain}`]) {
          currentAudit = cached[`audit_${domain}`];
          // Will be rendered after DOM loads
        }
      } catch {}
    }
  } catch {
    currentTab = null;
  }
}

    showState('results');
  }

In the tabChanged message listener, update to restore 
cached audit instead of always resetting to idle:

chrome.runtime.onMessage.addListener((message) => {
  if (message.action === 'tabChanged') {
    document.getElementById('domain-text').textContent = 
      message.domain || 'Unknown';
    currentAudit = null;
    
    // Check for cached audit for new domain
    if (message.domain) {
      chrome.storage.local.get(`audit_${message.domain}`, (data) => {
        const cached = data[`audit_${message.domain}`];
        if (cached) {
          currentAudit = cached;
          renderResults(cached);
          showState('results');
        } else {
          showState('idle');
        }
      });
    } else {
      showState('idle');
    }
  }
});
