function getCriticalIssues(botScore, schemaScore, answerScore) {
  const issues = [];
  if (botScore.citationBotsBlocked?.length > 0) {
    issues.push({ 
      severity: 'critical', 
      type: 'bot-blocking',
      text: `Citation bots blocked: ${botScore.citationBotsBlocked.join(', ')}` 
    });
  }
  if (!schemaScore.detected?.hasOrg) {
    issues.push({ 
      severity: 'high', 
      type: 'schema',
      text: 'Missing Organization schema markup' 
    });
  }
  if (answerScore.score < 8) {
    issues.push({ 
      severity: 'high', 
      type: 'answer-capsule',
      text: 'Opening content not structured as answer capsule' 
    });
  }
  return issues;
}
