  function splitCamelOrCompound(str) {
    // Insert space before sequences of capitals followed by lowercase
    // "apolloHospitals" → "apollo Hospitals"  
    // Try to split known compound words
    const withSpaces = str
      .replace(/([a-z])([A-Z])/g, '$1 $2')
      .replace(/([a-zA-Z])(\d)/g, '$1 $2');
    return withSpaces;
  }

  const base = domain.split('.')[0];
  const split = splitCamelOrCompound(base);
  
  const brandVariants = [
    // Standard: "zoho" → "Zoho"
    base.charAt(0).toUpperCase() + base.slice(1),
    // Split compound: "apollohospitals" → "Apollo Hospitals"  
    split.replace(/\b\w/g, c => c.toUpperCase()),
    // All caps for short names
    base.length <= 5 ? base.toUpperCase() : null,
    // With common suffixes stripped
    base.replace(/(app|hq|io|ai|hub)$/i, '')
        .replace(/\b\w/g, c => c.toUpperCase()),
  ].filter((v, i, arr) => v && arr.indexOf(v) === i); // unique, non-null

async function checkRedditPresence(domain) {
  const brandName = domain.split('.')[0];
  const brand = brandName.charAt(0).toUpperCase() + brandName.slice(1);
  
  try {
    const url = `https://www.reddit.com/search.json?q=${encodeURIComponent(brand)}&sort=top&t=year&limit=10&type=link`;
    const res = await fetch(url, {
      headers: { 'User-Agent': 'Mozilla/5.0' },
      signal: AbortSignal.timeout(8000),
    });
    if (!res.ok) return { found: false, mentionCount: 0, topSubreddits: [] };
    
    const data = await res.json();
    const posts = data?.data?.children ?? [];
    const relevant = posts.filter(p => 
      (p.data?.title || '').toLowerCase().includes(brandName.toLowerCase())
    );
    const subreddits = [...new Set(
      relevant.map(p => 'r/' + p.data?.subreddit).filter(Boolean)
    )].slice(0, 3);
    
    return {
      found: relevant.length > 0,
      mentionCount: relevant.length,
      topSubreddits: subreddits,
    };
  } catch {
    return { found: false, mentionCount: 0, topSubreddits: [] };
  }
}
