  app.post('/api/generate-aeo-files', requireAuth, async (req: AuthenticatedRequest, res) => {
    const userId = req.user?.uid;
    if (!userId) return res.status(401).json({ error: 'Unauthorized' });

    const userPlanRaw = req.body.userPlan || 'free';
    const serverPlan = await subscriptionService.getServerPlan(userId, userPlanRaw);
    const { getFeatureFlags } = await import('../shared/feature-flags');
    const flags = getFeatureFlags(serverPlan);

    const allowedPlans = ['pro', 'enterprise', 'lifetime'];
    if (!allowedPlans.includes(serverPlan)) {
      return res.status(403).json({
        error: 'AEO file generation requires a Pro plan or above.',
        code: 'PLAN_REQUIRED',
        upgrade: '/pricing',
      });
    }

    const { websiteUrl, companyName, industry, businessType,
            targetAudience, valueProposition, keywords = [],
            competitors = [], pingIndexNow = true } = req.body;

    if (!websiteUrl) {
      return res.status(400).json({ error: 'websiteUrl is required' });
    }

    try {
      const scrapeResult = await scrapeWebsiteContent(websiteUrl);
      const { extractContentSignals } = await import('./content-signal-service');
      const { generateAeoFiles, pingBingIndexNow } = await import('./aeo-file-generator');
      const archiver = (await import('archiver')).default;

      const signals = extractContentSignals(
        scrapeResult.html, scrapeResult.text, scrapeResult.sitemapUrls
      );

      const resolvedCompanyName = companyName ||
        new URL(websiteUrl).hostname.replace(/^www\./, '').split('.')[0];

      const files = await generateAeoFiles({
        websiteUrl,
        companyName: resolvedCompanyName,
        industry: industry || '',
        businessType: businessType || '',
        targetAudience: targetAudience || '',
        valueProposition: valueProposition || '',
        signals,
        sitemapUrls: scrapeResult.sitemapUrls,
        scrapedText: scrapeResult.text,
        keywords,
        competitors,
      });

      if (pingIndexNow) {
        pingBingIndexNow(websiteUrl, scrapeResult.sitemapUrls).catch(() => {});
      }

      res.setHeader('Content-Type', 'application/zip');
      res.setHeader('Content-Disposition',
        `attachment; filename="${resolvedCompanyName}-aeo-files.zip"`);

      const archive = archiver('zip', { zlib: { level: 9 } });
      archive.pipe(res);

      for (const [filename, content] of Object.entries(files)) {
        archive.append(content, { name: filename });
      }

      await archive.finalize();

    } catch (err: any) {
      console.error('❌ AEO file generation error:', err);
      if (!res.headersSent) {
        res.status(500).json({ error: 'File generation failed', details: err.message });
      }
    }
  });

const handleGenerateFiles = async () => {
  try {
    const token = await auth.currentUser?.getIdToken();
    const response = await fetch('/api/generate-aeo-files', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${token}`,
      },
      body: JSON.stringify({
        websiteUrl: auditData.websiteUrl,
        companyName: auditData.companyIntelligence?.companyName,
        industry: auditData.companyIntelligence?.industry,
        businessType: auditData.companyIntelligence?.businessType,
        targetAudience: auditData.companyIntelligence?.targetAudience,
        valueProposition: auditData.companyIntelligence?.valueProposition,
        keywords: auditData.companyIntelligence?.selectedKeywords || [],
        competitors: auditData.companyIntelligence?.confirmedCompetitors || [],
        userPlan: userPlan,
      }),
    });

    if (response.status === 403) {
      toast({ 
        title: 'Upgrade required',
        description: 'AEO file generation requires Pro plan or above.' 
      });
      return;
    }
    if (!response.ok) throw new Error('Generation failed');

    const blob = await response.blob();
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${auditData.companyIntelligence?.companyName || 'site'}-aeo-files.zip`;
    a.click();
    window.URL.revokeObjectURL(url);
  } catch (err) {
    toast({ 
      title: 'Error', 
      description: 'Could not generate files. Try again.' 
    });
  }
};
