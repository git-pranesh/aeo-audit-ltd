// API endpoint for audit
app.post('/api/audit', async (req, res) => {
  try {
    const { websiteUrl } = req.body;
    
    // Validate URL
    if (!websiteUrl || !validator.isURL(websiteUrl, { protocols: ['http', 'https'] })) {
      return res.status(400).json({ 
        error: 'Please provide a valid website URL (include http:// or https://)' 
      });
    }

    // Clean URL
    const cleanUrl = websiteUrl.trim();
    
    // Prepare prompt with actual URL
    const promptWithUrl = MASTER_PROMPT.replace('{WEBSITE_URL}', cleanUrl);
    
    // Call Perplexity API
    const response = await axios.post(
      'https://api.perplexity.ai/chat/completions',
      {
        model: 'llama-3.1-sonar-large-128k-online',
        messages: [
          {
            role: 'system',
            content: 'You are an expert AI Search Engine Optimization analyst. Always respond with valid JSON format.'
          },
          {
            role: 'user', 
            content: promptWithUrl
          }
        ],
        max_tokens: 8000,
        temperature: 0.2,
        top_p: 0.9,
        return_citations: true,
        search_domain_filter: ['perplexity.ai'],
        return_images: false,
        return_related_questions: false,
        search_recency_filter: 'month',
        top_k: 0,
        stream: false,
        presence_penalty: 0,
        frequency_penalty: 1
      },
      {
        headers: {
          'Authorization': `Bearer ${process.env.PERPLEXITY_API_KEY}`,
          'Content-Type': 'application/json'
        },
        timeout: 60000 // 60 second timeout
      }
    );

    // Parse response
    let auditData;
    try {
      const content = response.data.choices[0].message.content;
      auditData = JSON.parse(content);
    } catch (parseError) {
      console.error('JSON Parse Error:', parseError);
      return res.status(500).json({ 
        error: 'Failed to parse AI response. Please try again.' 
      });
    }

    // Validate response structure
    if (!auditData.auditResults || !auditData.contentTopics) {
      return res.status(500).json({ 
        error: 'Invalid response structure from AI. Please try again.' 
      });
    }

    // Return successful response
    res.json({
      success: true,
      websiteUrl: cleanUrl,
      auditData: auditData,
      generatedAt: new Date().toISOString()
    });

  } catch (error) {
    console.error('Audit Error:', error);
    
    if (error.response?.status === 401) {
      return res.status(500).json({ 
        error: 'API authentication failed. Please check configuration.' 
      });
    }
    
    if (error.response?.status === 429) {
      return res.status(429).json({ 
        error: 'Rate limit exceeded. Please try again in a few minutes.' 
      });
    }
    
    if (error.code === 'ECONNABORTED') {
      return res.status(408).json({ 
        error: 'Request timeout. The analysis is taking too long. Please try again.' 
      });
    }

    res.status(500).json({ 
      error: 'An unexpected error occurred. Please try again.' 
    });
  }
});

    <script>
        document.getElementById('auditForm').addEventListener('submit', async function(e) {
            e.preventDefault();
            
            const websiteUrl = document.getElementById('websiteUrl').value.trim();
            const submitBtn = document.getElementById('submitBtn');
            const loading = document.getElementById('loading');
            const results = document.getElementById('results');
            const errorMessage = document.getElementById('errorMessage');
            
            // Reset UI
            results.style.display = 'none';
            errorMessage.style.display = 'none';
            loading.style.display = 'block';
            submitBtn.disabled = true;
            submitBtn.textContent = 'Analyzing...';
            
            try {
                const response = await fetch('/api/audit', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({ websiteUrl })
                });
                
                const data = await response.json();
                
                if (!response.ok) {
                    throw new Error(data.error || 'An error occurred');
                }
                
                // Populate results
                populateResults(data.auditData);
                
                // Show results
                loading.style.display = 'none';
                results.style.display = 'block';
                
            } catch (error) {
                console.error('Error:', error);
                loading.style.display = 'none';
                errorMessage.textContent = error.message || 'An unexpected error occurred. Please try again.';
                errorMessage.style.display = 'block';
            } finally {
                submitBtn.disabled = false;
                submitBtn.textContent = 'Run AI Citation Audit';
            }
        });
        
        function populateResults(data) {
            // Populate audit results
            document.getElementById('overallScore').textContent = data.auditResults.overallScore;
            document.getElementById('letterGrade').textContent = data.auditResults.letterGrade;
            
            // Platform scores
            document.getElementById('chatgptScore').textContent = data.auditResults.platformScores.chatgpt;
            document.getElementById('perplexityScore').textContent = data.auditResults.platformScores.perplexity;
            document.getElementById('geminiScore').textContent = data.auditResults.platformScores.gemini;
            document.getElementById('claudeScore').textContent = data.auditResults.platformScores.claude;
            
            // Technical findings
            document.getElementById('domainAuthority').textContent = data.auditResults.technicalFindings.domainAuthority;
            document.getElementById('contentStructure').textContent = data.auditResults.technicalFindings.contentStructure;
            document.getElementById('schemaMarkup').textContent = data.auditResults.technicalFindings.schemaMarkup;
            document.getElementById('entityRecognition').textContent = data.auditResults.technicalFindings.entityRecognition;
            
            // Strengths
            const strengthsList = document.getElementById('strengthsList');
            strengthsList.innerHTML = '';
            data.auditResults.strengths.forEach(strength => {
                const li = document.createElement('li');
                li.textContent = strength;
                strengthsList.appendChild(li);
            });
            
            // Weaknesses
            const weaknessesList = document.getElementById('weaknessesList');
            weaknessesList.innerHTML = '';
            data.auditResults.weaknesses.forEach(weakness => {
                const li = document.createElement('li');
                li.textContent = weakness;
                weaknessesList.appendChild(li);
            });
            
            // Recommendations
            const recommendationsList = document.getElementById('recommendationsList');
            recommendationsList.innerHTML = '';
            data.recommendations.forEach(recommendation => {
                const li = document.createElement('li');
                li.textContent = recommendation;
                recommendationsList.appendChild(li);
            });
            
            // Content topics
            populateTopics('authoritativeTopics', data.contentTopics.authoritative);
            populateTopics('thoughtLeadershipTopics', data.contentTopics.thoughtLeadership);
            populateTopics('competitiveAdvantageTopics', data.contentTopics.competitiveAdvantage);
            populateTopics('emergingOpportunityTopics', data.contentTopics.emergingOpportunity);
        }
        
        function populateTopics(containerId, topics) {
            const container = document.getElementById(containerId);
            container.innerHTML = '';
            
            topics.forEach((topic, index) => {
                if (index < 10) { // Show first 10 topics
                    const topicDiv = document.createElement('div');
                    topicDiv.className = 'topic-item';
                    
                    topicDiv.innerHTML = `
                        <div class="topic-question">${topic.topic}</div>
                        <span class="topic-confidence">${topic.confidence}% confidence</span>
                        <div class="topic-justification">${topic.justification}</div>
                        <div class="topic-platforms">Platforms: ${topic.platforms.join(', ')}</div>
                    `;
                    
                    container.appendChild(topicDiv);
                }
            });
        }
