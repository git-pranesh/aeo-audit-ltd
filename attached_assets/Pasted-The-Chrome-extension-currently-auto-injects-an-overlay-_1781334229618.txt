2. Create extension/popup.js — this requests results from the 
   content script running in the current tab:

document.addEventListener('DOMContentLoaded', async () => {
  const loading = document.getElementById('loading');
  const results = document.getElementById('results');
  const errorDiv = document.getElementById('error');

  try {
    // Get current tab
    const [tab] = await chrome.tabs.query({ 
      active: true, currentWindow: true 
    });

    if (!tab?.id || !tab?.url || 
        tab.url.startsWith('chrome://') || 
        tab.url.startsWith('chrome-extension://')) {
      loading.style.display = 'none';
      errorDiv.style.display = 'block';
      return;
    }

    // Set domain
    const domain = new URL(tab.url).hostname.replace('www.', '');
    document.getElementById('domain-text').textContent = domain;

    // Run audit in the tab context
    const response = await chrome.tabs.sendMessage(tab.id, { 
      action: 'runAudit' 
    });

    if (!response || !response.audit) {
      throw new Error('No audit response');
    }

    renderResults(response.audit, tab.url);
  } catch (err) {
    loading.style.display = 'none';
    errorDiv.style.display = 'block';
  }
});

function renderResults(audit, pageUrl) {
  const loading = document.getElementById('loading');
  const results = document.getElementById('results');

  loading.style.display = 'none';
  results.style.display = 'block';

  const { score, grade, modules, criticalIssues } = audit;

  // Color logic
  const color = score >= 65 ? '#22c55e' : 
                score >= 40 ? '#f59e0b' : '#ef4444';

  // Score + grade
  const scoreEl = document.getElementById('score-num');
  const gradeEl = document.getElementById('grade-text');
  scoreEl.textContent = score;
  scoreEl.style.color = color;
  gradeEl.textContent = grade;
  gradeEl.style.color = color;

  // Progress bar
  const fill = document.getElementById('progress-fill');
  fill.style.width = score + '%';
  fill.style.background = color;

  // Critical issues banner
  if (criticalIssues && criticalIssues.length > 0) {
    document.getElementById('critical-banner').style.display = 'block';
    const itemsEl = document.getElementById('critical-items');
    itemsEl.innerHTML = criticalIssues.slice(0, 3).map(i => 
      `<div class="critical-item">• ${i.text}</div>`
    ).join('');
  }

  // Module grid
  const modulesEl = document.getElementById('modules-grid');
  const moduleList = Object.values(modules);
  modulesEl.innerHTML = moduleList.map(m => {
    const pct = m.score / m.max;
    const mc = pct >= 0.65 ? '#22c55e' : 
               pct >= 0.4 ? '#f59e0b' : '#ef4444';
    return `
      <div class="module">
        <div class="module-label">${m.label}</div>
        <div class="module-score" style="color:${mc}">
          ${m.score}<span style="font-size:10px;color:#555;font-weight:400">/${m.max}</span>
        </div>
        <div class="module-bar">
          <div class="module-fill" 
               style="width:${Math.round(pct*100)}%;background:${mc}">
          </div>
        </div>
      </div>`;
  }).join('');

  // Full audit link
  const btn = document.getElementById('full-audit-btn');
  btn.href = `https://citationrocket.com/?url=${encodeURIComponent(pageUrl)}`;
}

3. Update extension/content.js — remove the auto-inject UI code 
   and replace with a message listener:

At the BOTTOM of content.js, REMOVE the entire buildOverlay function 
and the final runQuickAudit().then(buildOverlay).catch(console.error) line.

REPLACE with this message listener:

chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
  if (message.action === 'runAudit') {
    runQuickAudit()
      .then(audit => sendResponse({ audit }))
      .catch(err => sendResponse({ audit: null, error: err.message }));
    return true; // keep channel open for async response
  }
});
