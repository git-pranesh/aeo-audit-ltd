import { useState, useEffect } from 'react';
import { TrendingUp, BarChart3 } from 'lucide-react';
import { Card, CardContent } from '@/components/ui/card';
import { scoreHistoryDb } from '@/lib/firestore-utils';
import type { ScoreHistoryEntry } from '@/lib/firestore-utils';
import { LineChart, Line, XAxis, Tooltip, ResponsiveContainer } from 'recharts';

interface ScoreHistoryChartProps {
  auditId: string;
}

export default function ScoreHistoryChart({ auditId }: ScoreHistoryChartProps) {
  const [history, setHistory] = useState<ScoreHistoryEntry[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    scoreHistoryDb.getScoreHistory(auditId)
      .then(setHistory)
      .catch(console.error)
      .finally(() => setLoading(false));
  }, [auditId]);

  if (loading) return null;

  const delta = history.length >= 2
    ? history[history.length - 1].overallScore - history[0].overallScore
    : null;

  return (
    <Card className="bg-card border border-border shadow-lg fade-in-up">
      <CardContent className="p-6">
        <div className="flex items-center gap-2 mb-4">
          <div className="w-8 h-8 bg-primary/20 rounded-lg flex items-center justify-center">
            <BarChart3 className="h-4 w-4 text-primary" />
          </div>
          <h3 className="text-lg font-semibold">Score History</h3>
        </div>

        {history.length < 2 ? (
          <div className="text-center py-6">
            <TrendingUp className="h-8 w-8 text-muted-foreground mx-auto mb-3 opacity-40" />
            <p className="text-sm text-muted-foreground">Run a re-audit to start tracking your score over time</p>
          </div>
        ) : (
          <>
            <ResponsiveContainer width="100%" height={160}>
              <LineChart data={history} margin={{ top: 8, right: 8, bottom: 8, left: 8 }}>
                <XAxis
                  dataKey="runAt"
                  tickFormatter={(val) => new Date(val).toLocaleDateString('en-US', { month: 'short', day: 'numeric' })}
                  tick={{ fontSize: 11, fill: 'hsl(var(--muted-foreground))' }}
                  axisLine={false}
                  tickLine={false}
                />
                <Tooltip
                  formatter={(val: number) => [`${val}/100`, 'Score']}
                  labelFormatter={(label) => new Date(label).toLocaleDateString('en-US', { month: 'long', day: 'numeric', year: 'numeric' })}
                  contentStyle={{ fontSize: 12 }}
                />
                <Line
                  type="monotone"
                  dataKey="overallScore"
                  stroke="hsl(var(--primary))"
                  strokeWidth={2}
                  dot={{ r: 4, fill: 'hsl(var(--primary))', strokeWidth: 0 }}
                  activeDot={{ r: 6 }}
                />
              </LineChart>
            </ResponsiveContainer>
            {delta !== null && (
              <p className={`text-sm text-center mt-2 font-medium ${delta > 0 ? 'text-green-600' : delta < 0 ? 'text-red-500' : 'text-muted-foreground'}`}>
                {delta > 0 ? `▲ +${delta} points since first audit` : delta < 0 ? `▼ ${Math.abs(delta)} points since first audit` : '→ No change since first audit'}
              </p>
            )}
          </>
        )}
      </CardContent>
    </Card>
  );
}
