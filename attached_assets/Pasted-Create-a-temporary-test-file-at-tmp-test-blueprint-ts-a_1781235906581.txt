import { runBlueprintAudit } from './server/blueprint-orchestrator';

async function test() {
  console.log('Testing blueprint orchestrator...');
  
  const sampleHtml = `
    <html>
      <head>
        <script type="application/ld+json">
          {"@type":"Organization","name":"TestCo","url":"https://testco.com","sameAs":["https://linkedin.com/company/testco"]}
        </script>
      </head>
      <body>
        <h1>TestCo helps SaaS companies grow faster</h1>
        <p>TestCo is a B2B SaaS platform that reduces churn by 40%. Founded in 2020, we serve 500+ customers.</p>
        <h2>How does TestCo work?</h2>
        <p>TestCo analyzes your customer data using machine learning to predict churn before it happens. Our system processes over 1 billion data points monthly to generate actionable insights for your team.</p>
        <h2>What makes TestCo different?</h2>
        <p>Unlike competitors, TestCo provides real-time alerts with 89% accuracy according to a 2024 study of 200 SaaS companies.</p>
        <a href="https://example.com">External link</a>
        <a href="https://privacy.com/policy">Privacy Policy</a>
      </body>
    </html>
  `;

  try {
    const result = await runBlueprintAudit({
      websiteUrl: 'https://testco.com',
      html: sampleHtml,
      text: sampleHtml.replace(/<[^>]+>/g, ' '),
      sitemapUrls: ['https://testco.com/blog/post-1', 'https://testco.com/faq', 'https://testco.com/tutorials/getting-started'],
      brandName: 'TestCo',
      companyName: 'TestCo',
      industry: 'SaaS',
      primaryQuery: 'B2B SaaS churn reduction',
    });
    
    console.log('✅ Blueprint audit completed');
    console.log('  Raw score:', result.rawScore, '/ 150');
    console.log('  Normalized:', result.normalizedScore, '/ 100');
    console.log('  Answer Capsule:', result.answerCapsule.answerCapsuleScore, '/ 25');
    console.log('  E-E-A-T:', result.eeat.eeAtScore, '/ 35');
    console.log('  Bot Access:', result.botAccess.botAccessScore, '/ 10');
    console.log('  Schema:', result.schemaCompleteness.schemaCompletenessScore, '/ 20');
    console.log('  Fan-Out:', result.fanOut.fanOutScore, '/ 10');
    console.log('  Freshness:', result.freshness.freshnessScore, '/ 15');
    console.log('  Critical gaps:', result.criticalGaps.length);
    console.log('  Quick wins:', result.quickWins.length);
    
    if (result.criticalGaps.length > 0) {
      console.log('  Top gap:', result.criticalGaps[0].signal, '(+' + result.criticalGaps[0].estimatedScoreGain + ' pts)');
    }
  } catch (err) {
    console.log('❌ Blueprint failed:', err.message);
    console.log(err.stack);
  }
}

test().catch(console.error);
