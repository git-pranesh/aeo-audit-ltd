FIX 2 — G2 profile check (try multiple URL patterns):
In extension/content.js, find the checkG2Profile function 
and replace it entirely with:

async function checkG2Profile(domain) {
  const slug = domain.split('.')[0].toLowerCase().replace(/[^a-z0-9]/g, '-');
  
  // Try multiple G2 URL patterns — G2 has inconsistent slugs
  const urlsToTry = [
    `https://www.g2.com/products/${slug}/reviews`,
    `https://www.g2.com/sellers/${slug}`,
    `https://www.g2.com/sellers/${slug}-corporation`,
    `https://www.g2.com/sellers/${slug}-inc`,
  ];
  
  for (const url of urlsToTry) {
    try {
      const res = await fetch(url, {
        method: 'HEAD',
        signal: AbortSignal.timeout(4000),
        redirect: 'follow',
      });
      if (res.ok && res.status < 400) {
        return { found: true, url };
      }
    } catch {
      continue;
    }
  }
  return { found: false, url: urlsToTry[0] };
}

async function checkWikidata(domain) {
  // Try multiple brand name variations
  const base = domain.split('.')[0];
  const brandVariants = [
    // Original capitalized: "zoho" → "Zoho"
    base.charAt(0).toUpperCase() + base.slice(1),
    // All caps for short names
    base.length <= 4 ? base.toUpperCase() : null,
    // Title case with spaces for hyphenated: "answer-the-public" → "Answer The Public"  
    base.replace(/-/g, ' ').replace(/\b\w/g, c => c.toUpperCase()),
  ].filter(Boolean);
  
  for (const brand of brandVariants) {
    try {
      const url = `https://www.wikidata.org/w/api.php?action=wbsearchentities&search=${encodeURIComponent(brand)}&language=en&limit=3&format=json&origin=*&type=item`;
      const res = await fetch(url, { signal: AbortSignal.timeout(6000) });
      if (!res.ok) continue;
      const data = await res.json();
      const results = data?.search ?? [];
      // Check if any result name closely matches our brand
      const match = results.find(r => {
        const label = (r.label || '').toLowerCase();
        const b = brand.toLowerCase();
        return label === b || label.startsWith(b) || b.startsWith(label);
      });
      if (match) {
        return { found: true, qNumber: match.id, brandGuess: brand };
      }
    } catch {
      continue;
    }
  }
  return { found: false, qNumber: null, brandGuess: brandVariants[0] };
}
