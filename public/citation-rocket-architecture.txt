import dotenv from "dotenv";
dotenv.config();

import express, { type Request, Response, NextFunction } from "express";
import { registerRoutes } from "./routes";
import { setupVite, serveStatic, log } from "./vite";

const app = express();
app.set('trust proxy', 1); // Trust first proxy for rate limiting

// IMPORTANT: Register webhook text body parser BEFORE global JSON parser
// This ensures webhook endpoints receive raw body for cryptographic signature verification
// The text parser is applied specifically to webhook paths first
app.use('/api/webhooks/dodo', express.text({ type: '*/*' }));

// Global JSON parser for all other endpoints
app.use(express.json());
app.use(express.urlencoded({ extended: false }));

// COOP header required for Firebase Auth popup to communicate back
app.use((req, res, next) => {
  res.setHeader('Cross-Origin-Opener-Policy', 'same-origin-allow-popups');
  next();
});

app.use((req, res, next) => {
  const start = Date.now();
  const path = req.path;
  let capturedJsonResponse: Record<string, any> | undefined = undefined;

  const originalResJson = res.json;
  res.json = function (bodyJson, ...args) {
    capturedJsonResponse = bodyJson;
    return originalResJson.apply(res, [bodyJson, ...args]);
  };

  res.on("finish", () => {
    const duration = Date.now() - start;
    if (path.startsWith("/api")) {
      let logLine = `${req.method} ${path} ${res.statusCode} in ${duration}ms`;
      if (capturedJsonResponse) {
        logLine += ` :: ${JSON.stringify(capturedJsonResponse)}`;
      }

      if (logLine.length > 80) {
        logLine = logLine.slice(0, 79) + "…";
      }

      log(logLine);
    }
  });

  next();
});

(async () => {
  const server = await registerRoutes(app);

  app.use((err: any, _req: Request, res: Response, _next: NextFunction) => {
    const status = err.status || err.statusCode || 500;
    const message = err.message || "Internal Server Error";

    res.status(status).json({ message });
    throw err;
  });

  // importantly only setup vite in development and after
  // setting up all the other routes so the catch-all route
  // doesn't interfere with the other routes
  if (app.get("env") === "development") {
    await setupVite(app, server);
  } else {
    serveStatic(app);
  }

  // ALWAYS serve the app on the port specified in the environment variable PORT
  // Other ports are firewalled. Default to 5001 if not specified.
  // this serves both the API and the client.
  // It is the only port that is not firewalled.
  const port = parseInt(process.env.PORT || '5001', 10);
  server.listen({
    port,
    host: "0.0.0.0",
    reusePort: true,
  }, () => {
    log(`serving on port ${port}`);
  });
})();

 * - Tutorial/how-to content count (ChatGPT)
 * - FAQ sections (ChatGPT)
 * - Research citations/data points (Perplexity)
 * - Schema markup presence (Gemini)
 * - Case studies/frameworks (Claude)
 */

export interface ContentSignals {
  // ChatGPT signals (practical, how-to content)
  tutorialCount: number;
  howToGuideCount: number;
  faqSectionCount: number;
  stepByStepListCount: number;
  
  // Perplexity signals (research depth, citations)
  externalLinkCount: number;
  dataPointCount: number;
  statisticMentionCount: number;
  researchTermCount: number;
  
  // Gemini signals (structure, SEO, multimedia)
  schemaTypeCount: number;
  headingHierarchyScore: number;
  hasStructuredData: boolean;
  imageCount: number;
  
  // Claude signals (analytical depth, frameworks)
  caseStudyCount: number;
  frameworkMentionCount: number;
  longFormContentScore: number;
  nuancedTermCount: number;
  
  // General signals
  blogPostCount: number;
  documentationPageCount: number;
  totalPageCount: number;
  contentDepthScore: number;
  
  // Candidate URLs for verification (NEW)
  candidateUrls?: {
    tutorials: string[];
    caseStudies: string[];
    faqs: string[];
    research: string[];
    frameworks: string[];
  };
}

export interface PlatformScores {
  chatgpt: number;
  perplexity: number;
  gemini: number;
  claude: number;
}

export interface PlatformRationale {
  chatgpt: string;
  perplexity: string;
  gemini: string;
  claude: string;
}

export interface ScoringResult {
  overallScore: number;
  letterGrade: string;
  platformScores: PlatformScores;
  platformRationale: PlatformRationale;
  signals: ContentSignals;
}

/**
 * Extract content signals from scraped HTML and text content
 */
export function extractContentSignals(
  htmlContent: string,
  textContent: string,
  sitemapUrls: string[] = [],
  companyIntelligence: any = {}
): ContentSignals {
  const html = htmlContent.toLowerCase();
  const text = textContent.toLowerCase();
  
  // ChatGPT signals - practical how-to content
  // SITEMAP-FIRST: Count URLs in sitemap that indicate tutorials/how-to content
  const tutorialUrls = sitemapUrls.filter(url => 
    /\/tutorial|\/how-to|\/guide|\/learn|\/getting-started|\/walkthrough|\/step-by-step/i.test(url)
  );
  const tutorialUrlCount = tutorialUrls.length;
  
  // Also count text mentions as secondary signal
  const tutorialPatterns = [
    /how to /gi, /step[- ]by[- ]step/gi, /tutorial/gi, /guide to/gi,
    /getting started/gi, /beginners? guide/gi, /walkthrough/gi
  ];
  const tutorialTextMentions = tutorialPatterns.reduce((count, pattern) => 
    count + (text.match(pattern) || []).length, 0);
  
  // Use sitemap count if available, otherwise fall back to text mentions (divided to avoid inflation)
  const tutorialCount = tutorialUrlCount > 0 ? tutorialUrlCount : Math.floor(tutorialTextMentions / 3);
  
  const howToGuideCount = (text.match(/how to [a-z]/gi) || []).length;
  
  // SITEMAP-FIRST: Count FAQ pages from sitemap
  const faqUrls = sitemapUrls.filter(url => {
    const normalizedUrl = url.toLowerCase().replace(/[?#].*$/, '');
    // Strong FAQ patterns - always count these
    const strongFaqPattern = /\/faq|\/faqs\/|\/frequently-asked|\/questions\/|\/answers\/|\/knowledge-base|\/kb\//i.test(normalizedUrl);
    // Allow FAQ pages on any subdomain if URL explicitly contains "faq"
    const explicitFaq = /faq/i.test(normalizedUrl);
    return strongFaqPattern || explicitFaq;
  });
  const faqUrlCount = faqUrls.length;
  
  const faqPatterns = [
    /<script type="application\/ld\+json"[^>]*>[\s\S]*?faqpage[\s\S]*?<\/script>/gi,
    /frequently asked questions/gi, /faqs?[\s:]/gi, /<details/gi
  ];
  const faqTextMentions = faqPatterns.reduce((count, pattern) => 
    count + (html.match(pattern) || []).length, 0);
  
  // HYBRID: Use maximum of URL-based and text-based detection
  const faqSectionCount = Math.max(faqUrlCount, faqTextMentions);
  
  const stepByStepListCount = (html.match(/<ol[^>]*>[\s\S]*?<\/ol>/gi) || []).length +
    (text.match(/step \d+[:.]/gi) || []).length;
  
  // Perplexity signals - research depth
  // SITEMAP-FIRST: Count research/data pages from sitemap
  const researchUrls = sitemapUrls.filter(url => 
    /\/research|\/report|\/data|\/insight|\/statistic|\/study|\/whitepaper|\/ebook|\/benchmark/i.test(url)
  );
  const researchUrlCount = researchUrls.length;
  
  const externalLinkCount = (html.match(/href="https?:\/\/(?!.*?(facebook|twitter|instagram|linkedin|youtube))[^"]+"/gi) || []).length;
  
  const dataPointPatterns = [
    /\d+%/g, /\$[\d,]+/g, /\d+x/gi, /\d+ (million|billion|thousand)/gi,
    /increased by \d+/gi, /reduced by \d+/gi, /grew \d+/gi
  ];
  const dataPointTextCount = dataPointPatterns.reduce((count, pattern) => 
    count + (text.match(pattern) || []).length, 0);
  
  // Boost data points if research URLs exist in sitemap
  const dataPointCount = dataPointTextCount + (researchUrlCount * 5);
  
  const statisticMentionCount = (text.match(/according to|research shows|study found|data indicates|survey|report/gi) || []).length;
  
  const researchTerms = [
    /peer[- ]reviewed/gi, /academic/gi, /published in/gi, /journal/gi,
    /methodology/gi, /findings/gi, /analysis shows/gi, /evidence/gi
  ];
  const researchTermCount = researchTerms.reduce((count, pattern) => 
    count + (text.match(pattern) || []).length, 0) + researchUrlCount;
  
  // Gemini signals - structure and SEO
  const schemaMatches = html.match(/<script type="application\/ld\+json"[^>]*>/gi) || [];
  const schemaTypeCount = schemaMatches.length;
  
  const h1Count = (html.match(/<h1/gi) || []).length;
  const h2Count = (html.match(/<h2/gi) || []).length;
  const h3Count = (html.match(/<h3/gi) || []).length;
  const headingHierarchyScore = Math.min(100, (h1Count > 0 ? 30 : 0) + 
    Math.min(40, h2Count * 8) + Math.min(30, h3Count * 5));
  
  const hasStructuredData = schemaTypeCount > 0;
  
  const imageCount = (html.match(/<img[^>]+>/gi) || []).length;
  
  // Claude signals - analytical depth
  // SITEMAP-FIRST: Count case study pages from sitemap
  // IMPORTANT: Exclude documentation subdomains that falsely match generic patterns
  const caseStudyUrls = sitemapUrls.filter(url => {
    const normalizedUrl = url.toLowerCase().replace(/[?#].*$/, ''); // Strip query/hash
    // Documentation subdomains should be excluded from /success/ and /customer/ patterns
    const isDocSubdomain = /^https?:\/\/(success|docs|help|support|learn|community)\./i.test(url);
    
    // Strong case study patterns - always count these
    const strongPattern = /\/case-stud|\/customer-stor|\/success-stor|\/testimonial|\/client-stor|\/stories\/|\/portfolio\//i.test(normalizedUrl);
    
    // Moderate patterns - only count on main domain or non-doc subdomains
    const moderatePattern = /\/customers?\/[a-z0-9-]+|\/success\/[a-z0-9-]+|\/results\/[a-z0-9-]+/i.test(normalizedUrl);
    
    return strongPattern || (moderatePattern && !isDocSubdomain);
  });
  const caseStudyUrlCount = caseStudyUrls.length;
  
  const caseStudyPatterns = [
    /case study/gi, /customer story/gi, /success story/gi,
    /client spotlight/gi, /use case/gi, /real[- ]world example/gi
  ];
  const caseStudyTextMentions = caseStudyPatterns.reduce((count, pattern) => 
    count + (text.match(pattern) || []).length, 0);
  
  // HYBRID: Use maximum of URL-based and text-based detection (avoids false positives zeroing out real signals)
  const textBasedCaseStudies = Math.floor(caseStudyTextMentions / 2);
  const caseStudyCount = Math.max(caseStudyUrlCount, textBasedCaseStudies);
  
  // SITEMAP-FIRST: Count framework/methodology pages from sitemap
  const frameworkUrls = sitemapUrls.filter(url => 
    /\/framework|\/methodology|\/approach|\/process|\/playbook|\/strategy|\/best-practice/i.test(url)
  );
  const frameworkUrlCount = frameworkUrls.length;
  
  const frameworkPatterns = [
    /framework/gi, /methodology/gi, /model for/gi, /approach to/gi,
    /strategy for/gi, /best practices/gi, /principles of/gi, /playbook/gi
  ];
  const frameworkTextMentions = frameworkPatterns.reduce((count, pattern) => 
    count + (text.match(pattern) || []).length, 0);
  
  // HYBRID: Use maximum of URL-based and text-based detection
  const textBasedFrameworks = Math.floor(frameworkTextMentions / 3);
  const frameworkMentionCount = Math.max(frameworkUrlCount, textBasedFrameworks);
  
  const wordCount = text.split(/\s+/).length;
  const longFormContentScore = Math.min(100, Math.floor(wordCount / 100));
  
  const nuancedTerms = [
    /however/gi, /on the other hand/gi, /nuanced/gi, /complex/gi,
    /trade[- ]off/gi, /considerations/gi, /depending on/gi, /context/gi,
    /it depends/gi, /caveat/gi, /exception/gi
  ];
  const nuancedTermCount = nuancedTerms.reduce((count, pattern) => 
    count + (text.match(pattern) || []).length, 0);
  
  // General signals from sitemap
  const blogPostCount = sitemapUrls.filter(url => 
    /\/blog\/|\/articles?\/|\/posts?\/|\/news\//i.test(url)
  ).length || Math.floor((text.match(/blog|article|post/gi) || []).length / 5);
  
  const documentationPageCount = sitemapUrls.filter(url => 
    /\/docs?\/|\/documentation\/|\/api\/|\/reference\/|\/guide\//i.test(url)
  ).length || (text.match(/documentation|api reference|developer guide/gi) || []).length;
  
  const totalPageCount = sitemapUrls.length || 10;
  
  const contentDepthScore = Math.min(100, 
    (blogPostCount * 2) + (documentationPageCount * 3) + 
    (tutorialCount * 2) + (caseStudyCount * 5));
  
  return {
    tutorialCount,
    howToGuideCount,
    faqSectionCount,
    stepByStepListCount,
    externalLinkCount,
    dataPointCount,
    statisticMentionCount,
    researchTermCount,
    schemaTypeCount,
    headingHierarchyScore,
    hasStructuredData,
    imageCount,
    caseStudyCount,
    frameworkMentionCount,
    longFormContentScore,
    nuancedTermCount,
    blogPostCount,
    documentationPageCount,
    totalPageCount,
    contentDepthScore,
    // Include candidate URLs for verification
    candidateUrls: {
      tutorials: tutorialUrls,
      caseStudies: caseStudyUrls,
      faqs: faqUrls,
      research: researchUrls,
      frameworks: frameworkUrls,
    }
  };
}

/**
 * Calculate platform scores based on extracted content signals
 * Scores range from 50-95 based on actual content features
 * 
 * CRITICAL: Even with sparse/no content, scores MUST be distinct across platforms
 */
export function calculatePlatformScores(signals: ContentSignals): ScoringResult {
  // Calculate base scores from content signals
  // ChatGPT score - practical how-to content
  const chatgptRaw = 
    Math.min(40, signals.tutorialCount * 4) +
    Math.min(25, signals.faqSectionCount * 5) +
    Math.min(20, signals.howToGuideCount * 2) +
    Math.min(15, signals.stepByStepListCount * 3);
  
  // Perplexity score - research depth and citations
  const perplexityRaw = 
    Math.min(30, signals.externalLinkCount * 0.5) +
    Math.min(30, signals.dataPointCount * 1.5) +
    Math.min(25, signals.statisticMentionCount * 3) +
    Math.min(15, signals.researchTermCount * 3);
  
  // Gemini score - structure and SEO
  const geminiRaw = 
    Math.min(35, signals.schemaTypeCount * 7) +
    (signals.headingHierarchyScore * 0.35) +
    Math.min(20, signals.imageCount * 0.5) +
    (signals.hasStructuredData ? 10 : 0);
  
  // Claude score - analytical depth and frameworks
  const claudeRaw = 
    Math.min(35, signals.caseStudyCount * 7) +
    Math.min(30, signals.frameworkMentionCount * 3) +
    Math.min(20, signals.longFormContentScore * 0.2) +
    Math.min(15, signals.nuancedTermCount * 1.5);
  
  // Calculate total signal strength to determine base level
  const totalSignalStrength = chatgptRaw + perplexityRaw + geminiRaw + claudeRaw;
  
  // Determine base score tier based on overall content strength
  // This ensures even sparse content gets a reasonable base
  let baseTier = 50;
  if (totalSignalStrength > 200) baseTier = 70;
  else if (totalSignalStrength > 100) baseTier = 62;
  else if (totalSignalStrength > 50) baseTier = 56;
  else if (totalSignalStrength > 20) baseTier = 52;
  // Sparse content stays at 50
  
  // Calculate individual scores with content-aware adjustment
  const chatgptScore = Math.min(95, baseTier + Math.floor(chatgptRaw * 0.3));
  const perplexityScore = Math.min(95, baseTier + Math.floor(perplexityRaw * 0.3));
  const geminiScore = Math.min(95, baseTier + Math.floor(geminiRaw * 0.3));
  const claudeScore = Math.min(95, baseTier + Math.floor(claudeRaw * 0.3));
  
  // Create scored array for gap enforcement
  const scores = [
    { platform: 'chatgpt' as const, score: chatgptScore, rawScore: chatgptRaw },
    { platform: 'perplexity' as const, score: perplexityScore, rawScore: perplexityRaw },
    { platform: 'gemini' as const, score: geminiScore, rawScore: geminiRaw },
    { platform: 'claude' as const, score: claudeScore, rawScore: claudeRaw }
  ];
  
  // Sort by score descending, then by raw score for tiebreaker
  scores.sort((a, b) => {
    if (b.score !== a.score) return b.score - a.score;
    return b.rawScore - a.rawScore;
  });
  
  // CRITICAL: Guarantee distinct scores using fixed offset distribution
  // This ensures different scores even when content signals are identical or zero
  
  // Define fixed score offsets to guarantee distinct scores
  // Order: 1st place = +0, 2nd = -5, 3rd = -9, 4th = -14 (minimum 4-5 point gaps)
  const scoreOffsets = [0, -5, -9, -14];
  
  // Calculate the highest score (will be used as anchor)
  const highestCalculatedScore = Math.max(
    scores[0].score,
    64  // Minimum anchor to allow 14-point spread above 50
  );
  
  // Assign final scores using fixed offsets from the anchor
  const finalScores: { platform: 'chatgpt' | 'perplexity' | 'gemini' | 'claude', score: number }[] = [];
  
  for (let i = 0; i < scores.length; i++) {
    const offsetScore = highestCalculatedScore + scoreOffsets[i];
    // Clamp to 50-95 range, but since we anchor at min 64, lowest will be 64-14=50
    const clampedScore = Math.min(95, Math.max(50, offsetScore));
    finalScores.push({ platform: scores[i].platform, score: clampedScore });
  }
  
  // Verify all scores are distinct (should always pass with fixed offsets)
  const uniqueScores = new Set(finalScores.map(s => s.score));
  if (uniqueScores.size < 4) {
    // Fallback: force completely deterministic distinct scores based on platform order
    console.warn('⚠️ Score collision detected, using fallback distribution');
    const fallbackBase = Math.max(64, Math.min(81, highestCalculatedScore));
    finalScores[0].score = fallbackBase;
    finalScores[1].score = fallbackBase - 5;
    finalScores[2].score = fallbackBase - 9;
    finalScores[3].score = fallbackBase - 14;
  }
  
  const platformScores: PlatformScores = {
    chatgpt: finalScores.find(s => s.platform === 'chatgpt')!.score,
    perplexity: finalScores.find(s => s.platform === 'perplexity')!.score,
    gemini: finalScores.find(s => s.platform === 'gemini')!.score,
    claude: finalScores.find(s => s.platform === 'claude')!.score
  };
  
  // Final verification: log if scores are not distinct (should never happen)
  const finalUniqueScores = new Set([platformScores.chatgpt, platformScores.perplexity, platformScores.gemini, platformScores.claude]);
  if (finalUniqueScores.size < 4) {
    console.error('❌ CRITICAL: Platform scores are not distinct:', platformScores);
  }
  
  const overallScore = Math.round(
    (platformScores.chatgpt + platformScores.perplexity + 
     platformScores.gemini + platformScores.claude) / 4
  );
  
  let letterGrade = 'F';
  if (overallScore >= 90) letterGrade = 'A';
  else if (overallScore >= 80) letterGrade = 'B+';
  else if (overallScore >= 70) letterGrade = 'B';
  else if (overallScore >= 60) letterGrade = 'C';
  else if (overallScore >= 50) letterGrade = 'D';
  
  // Generate rationale based on actual signals
  const platformRationale: PlatformRationale = {
    chatgpt: generateChatGPTRationale(signals, platformScores.chatgpt),
    perplexity: generatePerplexityRationale(signals, platformScores.perplexity),
    gemini: generateGeminiRationale(signals, platformScores.gemini),
    claude: generateClaudeRationale(signals, platformScores.claude)
  };
  
  return {
    overallScore,
    letterGrade,
    platformScores,
    platformRationale,
    signals
  };
}

function generateChatGPTRationale(signals: ContentSignals, score: number): string {
  const tutorialStrength = signals.tutorialCount >= 10 ? 'strong' : signals.tutorialCount >= 5 ? 'moderate' : 'limited';
  const faqPresence = signals.faqSectionCount >= 3 ? 'with FAQ sections' : '';
  
  if (score >= 80) {
    return `Strong practical content with ${signals.tutorialCount} tutorials and ${signals.howToGuideCount} how-to guides ${faqPresence}`;
  } else if (score >= 65) {
    return `Moderate how-to coverage with ${signals.tutorialCount} tutorials; could expand step-by-step guides`;
  } else {
    return `Limited practical content: ${signals.tutorialCount} tutorials found; ChatGPT prefers more how-to guides and FAQs`;
  }
}

function generatePerplexityRationale(signals: ContentSignals, score: number): string {
  if (score >= 80) {
    return `Research-rich content with ${signals.dataPointCount} data points, ${signals.statisticMentionCount} cited statistics, and ${signals.externalLinkCount} external references`;
  } else if (score >= 65) {
    return `Moderate research depth with ${signals.dataPointCount} data points; adding more citations would improve Perplexity visibility`;
  } else {
    return `Limited research backing: ${signals.statisticMentionCount} statistics found; Perplexity prefers data-driven, citation-heavy content`;
  }
}

function generateGeminiRationale(signals: ContentSignals, score: number): string {
  const schemaStatus = signals.schemaTypeCount > 0 ? `${signals.schemaTypeCount} schema types` : 'no structured data';
  
  if (score >= 80) {
    return `Well-structured with ${schemaStatus}, clear heading hierarchy (score: ${signals.headingHierarchyScore}), and ${signals.imageCount} images`;
  } else if (score >= 65) {
    return `Moderate structure with ${schemaStatus}; improving heading hierarchy and adding more schema markup would help`;
  } else {
    return `Weak structure: ${schemaStatus}, heading score ${signals.headingHierarchyScore}/100; Gemini favors well-organized, schema-rich content`;
  }
}

function generateClaudeRationale(signals: ContentSignals, score: number): string {
  if (score >= 80) {
    return `Strong analytical depth with ${signals.caseStudyCount} case studies, ${signals.frameworkMentionCount} frameworks, and nuanced long-form content`;
  } else if (score >= 65) {
    return `Moderate analytical content with ${signals.caseStudyCount} case studies; adding more frameworks and in-depth analysis would improve Claude visibility`;
  } else {
    return `Limited analytical depth: ${signals.caseStudyCount} case studies, ${signals.frameworkMentionCount} frameworks; Claude prefers nuanced, long-form analysis`;
  }
}

/**
 * Generate content-based strengths based on actual signals
 */
export function generateSignalBasedStrengths(
  signals: ContentSignals,
  companyName: string,
  competitors: string[] = []
): string[] {
  const strengths: string[] = [];
  
  if (signals.tutorialCount >= 10) {
    strengths.push(`${companyName} has ${signals.tutorialCount} tutorial/how-to pages, providing strong practical value for users seeking step-by-step guidance`);
  }
  
  if (signals.caseStudyCount >= 5) {
    strengths.push(`${companyName} features ${signals.caseStudyCount} case studies demonstrating real-world success, strong for building credibility`);
  }
  
  if (signals.schemaTypeCount >= 3) {
    strengths.push(`${companyName} implements ${signals.schemaTypeCount} schema markup types, enhancing discoverability in AI and search platforms`);
  }
  
  if (signals.dataPointCount >= 20) {
    strengths.push(`${companyName}'s content includes ${signals.dataPointCount}+ data points and statistics, establishing research credibility`);
  }
  
  if (signals.faqSectionCount >= 3) {
    strengths.push(`${companyName} has ${signals.faqSectionCount} FAQ sections, directly addressing common user questions that AI platforms frequently cite`);
  }
  
  if (signals.blogPostCount >= 50) {
    strengths.push(`${companyName} maintains an active blog with ${signals.blogPostCount}+ posts, providing broad topic coverage`);
  }
  
  if (signals.documentationPageCount >= 20) {
    strengths.push(`${companyName} offers comprehensive documentation (${signals.documentationPageCount}+ pages), valuable for technical audiences`);
  }
  
  // If no specific strengths found, provide honest assessment instead of generic filler
  if (strengths.length === 0) {
    // Return empty array - the UI should handle this case appropriately
    // rather than showing a misleading "strength"
    return [];
  }
  
  return strengths.slice(0, 5);
}

/**
 * Generate content-based weaknesses based on actual signals
 * Uses honest industry benchmarks instead of fake competitor data
 */
export function generateSignalBasedWeaknesses(
  signals: ContentSignals,
  companyName: string,
  competitors: string[] = []
): string[] {
  const weaknesses: string[] = [];
  
  if (signals.tutorialCount < 5) {
    weaknesses.push(`Your site has ${signals.tutorialCount} tutorials (industry benchmark: 15+)—content gap for practical how-to content that ChatGPT prefers`);
  }
  
  if (signals.caseStudyCount < 3) {
    weaknesses.push(`Your site has ${signals.caseStudyCount} case studies (industry benchmark: 8+)—limited social proof for AI platforms to cite`);
  }
  
  if (signals.schemaTypeCount < 2) {
    weaknesses.push(`Limited schema markup detected (${signals.schemaTypeCount} type${signals.schemaTypeCount === 1 ? '' : 's'}); implementing FAQ, HowTo, and Article schemas would improve AI discoverability`);
  }
  
  if (signals.dataPointCount < 10) {
    weaknesses.push(`Content lacks data backing (${signals.dataPointCount} data points found; benchmark: 25+); adding statistics and research citations would boost Perplexity visibility`);
  }
  
  if (signals.faqSectionCount < 2) {
    weaknesses.push(`Only ${signals.faqSectionCount} FAQ sections found (benchmark: 5+); adding structured FAQs would improve ChatGPT citation potential`);
  }
  
  if (signals.frameworkMentionCount < 3) {
    weaknesses.push(`Limited analytical frameworks (${signals.frameworkMentionCount} found; benchmark: 10+); Claude prefers content with clear methodologies and models`);
  }
  
  // Ensure at least one weakness
  if (weaknesses.length === 0) {
    weaknesses.push(`${companyName} has room to expand content depth across all AI platform preferences`);
  }
  
  return weaknesses.slice(0, 5);
}

/**
 * Generate signal-based recommendations with specific action items
 */
export function generateSignalBasedRecommendations(
  signals: ContentSignals,
  companyName: string,
  competitors: string[] = []
): string[] {
  const recommendations: string[] = [];
  
  // ChatGPT-focused recommendations (uses honest industry benchmarks)
  if (signals.tutorialCount < 10) {
    const target = Math.max(10, signals.tutorialCount + 7);
    recommendations.push(`Your site: ${signals.tutorialCount} tutorials | Industry benchmark: 15+ tutorials for high AI visibility | Action: Create ${target - signals.tutorialCount} step-by-step how-to guides`);
  }
  
  if (signals.faqSectionCount < 3) {
    recommendations.push(`Your site: ${signals.faqSectionCount} FAQ sections | Industry benchmark: 5+ FAQ pages | Action: Add 3-5 FAQ pages with structured markup for ChatGPT visibility`);
  }
  
  // Perplexity-focused recommendations
  if (signals.dataPointCount < 15) {
    recommendations.push(`Your site: ${signals.dataPointCount} data points | Industry benchmark: 25+ statistics per site | Action: Add research-backed statistics to ${Math.min(10, 15 - signals.dataPointCount)} existing articles`);
  }
  
  // Gemini-focused recommendations
  if (signals.schemaTypeCount < 3) {
    recommendations.push(`Your site: ${signals.schemaTypeCount} schema types | Industry benchmark: 5+ schema types | Action: Implement FAQ, HowTo, Article, and Product schemas across key pages`);
  }
  
  // Claude-focused recommendations
  if (signals.caseStudyCount < 5) {
    const target = Math.max(5, signals.caseStudyCount + 3);
    recommendations.push(`Your site: ${signals.caseStudyCount} case studies | Industry benchmark: 8+ case studies | Action: Create ${target - signals.caseStudyCount} detailed customer success stories`);
  }
  
  if (signals.frameworkMentionCount < 5) {
    recommendations.push(`Your site: ${signals.frameworkMentionCount} frameworks | Industry benchmark: 10+ methodology references | Action: Develop 3-5 proprietary methodology/framework articles for Claude visibility`);
  }
  
  // General recommendations
  if (signals.blogPostCount < 30) {
    recommendations.push(`Your site: ${signals.blogPostCount} blog posts | Industry benchmark: 50+ articles | Action: Increase publishing cadence to 4+ posts/month`);
  }
  
  return recommendations.slice(0, 7);
}

/**
 * Calculate score as percentage of benchmark (0-100%)
 * Capped at 100% - you can't score higher than benchmark
 */
function percentOfBenchmark(actual: number, benchmark: number): number {
  if (benchmark === 0) return 0;
  return Math.min(100, (actual / benchmark) * 100);
}

/**
 * Apply penalty curve for weak signals
 * Returns a multiplier 0-1 that gets harsher below benchmark thresholds
 */
function getPenaltyMultiplier(percentOfBenchmark: number): number {
  if (percentOfBenchmark >= 60) {
    // Normal scoring: 60-100% of benchmark = full credit
    return 1.0;
  } else if (percentOfBenchmark >= 30) {
    // Moderate penalty: 30-60% of benchmark = 70% credit
    return 0.7;
  } else if (percentOfBenchmark >= 10) {
    // Harsh penalty: 10-30% of benchmark = 40% credit
    return 0.4;
  } else {
    // Very harsh: <10% of benchmark = 15% credit
    return 0.15;
  }
}

  const tutorialPct = percentOfBenchmark(signals.tutorialCount, COMPETITOR_BENCHMARKS.tutorials);

/**
 * Calculate platform-specific score using weighted categories
 * Category scores are first normalized to 0-100 scale, then weighted
 */
function calculatePlatformScore(
  contentDepth: number,
  structureTechnical: number,
  researchCitations: number,
  authority: number,
  platform: 'chatgpt' | 'perplexity' | 'gemini' | 'claude'
): number {
  const weights = PLATFORM_WEIGHTS[platform];
  
  // Normalize category scores to 0-100 scale
  // contentDepth: 0-35 → 0-100
  // structureTechnical: 0-25 → 0-100
  // researchCitations: 0-25 → 0-100
  // authority: 0-15 → 0-100
  const normalizedContent = (contentDepth / 35) * 100;
  const normalizedStructure = (structureTechnical / 25) * 100;
  const normalizedResearch = (researchCitations / 25) * 100;
  const normalizedAuthority = (authority / 15) * 100;
  
  const weightedScore = 
    normalizedContent * weights.contentDepth +
    normalizedStructure * weights.structureTechnical +
    normalizedResearch * weights.researchCitations +
    normalizedAuthority * weights.authority;
  
  return weightedScore;
}

/**
 * Apply small tie-breakers to ensure distinct scores
 * Only ±1-2 points, applied AFTER true scores are calculated
 */
function ensureDistinctScores(scores: PlatformScores): PlatformScores {
  const platforms: ('chatgpt' | 'perplexity' | 'gemini' | 'claude')[] = 
    ['chatgpt', 'perplexity', 'gemini', 'claude'];
  
  // Sort by score descending
  const sorted = platforms.map(p => ({ platform: p, score: scores[p] }))
    .sort((a, b) => b.score - a.score);
  
  // Apply minimal adjustments to break ties (max ±2 points)
  const result: PlatformScores = { chatgpt: 0, perplexity: 0, gemini: 0, claude: 0 };
  const usedScores = new Set<number>();
  
  for (const item of sorted) {
    let finalScore = Math.round(item.score);
    
    // If score already used, adjust by -1, -2 until unique
    let offset = 0;
    while (usedScores.has(finalScore) && offset < 3) {
      offset++;
      finalScore = Math.max(0, Math.round(item.score) - offset);
    }
    
    usedScores.add(finalScore);
    result[item.platform] = finalScore;
  }
  
  return result;
}

/**
 * Calculate letter grade from overall score
 * Stricter grading aligned with professional SEO tools
 */
function calculateLetterGrade(score: number): string {
  if (score >= 90) return 'A+';
  if (score >= 85) return 'A';
  if (score >= 80) return 'A-';
  if (score >= 75) return 'B+';
  if (score >= 70) return 'B';
  if (score >= 65) return 'B-';
  if (score >= 60) return 'C+';
  if (score >= 55) return 'C';
  if (score >= 50) return 'C-';
  if (score >= 45) return 'D+';
  if (score >= 40) return 'D';
  if (score >= 35) return 'D-';
  return 'F';
}

  const authority = calculateAuthority(signals);
  
  // Total base score (sum of categories, max 100)

    claude: calculatePlatformScore(contentDepth, structureTechnical, researchCitations, authority, 'claude')
  };
  
  // Apply small tie-breakers to ensure distinct scores
  const platformScores = ensureDistinctScores(rawPlatformScores);
  
  // Overall score is the total base score (sum of category scores)
  // This ensures consistency between breakdown and overall score
  const overallScore = Math.round(totalBase);
  
  const letterGrade = calculateLetterGrade(overallScore);
  
  console.log('📊 Zero-Based Scoring Breakdown:', {

    rawPlatformScores: {
      chatgpt: rawPlatformScores.chatgpt.toFixed(1),
      perplexity: rawPlatformScores.perplexity.toFixed(1),
      gemini: rawPlatformScores.gemini.toFixed(1),

  return {

      structureTechnical: Math.round(structureTechnical),
      researchCitations: Math.round(researchCitations),
      authority: Math.round(authority),
      total: Math.round(totalBase)
    },

  };
}

// Helper function to extract domain name from URL
function extractDomain(url: string): string {
  try {
    const urlObj = new URL(url.startsWith('http') ? url : `https://${url}`);
    return urlObj.hostname.replace(/^www\./, '');
  } catch {
    // Fallback: extract domain-like pattern from string
    const match = url.match(/(?:https?:\/\/)?(?:www\.)?([^\/\s]+)/);
    return match ? match[1] : url;
  }
}

            sitemapUrls.push(...urlMatches.map((m: string) => m.replace(/<\/?loc>/gi, '')));
          } catch { /* skip inaccessible sitemaps */ }
        }
        return sitemapUrls;
      } catch {
        return [];
      }
    };
    
    // Check common subdomains in parallel using GET (HEAD often blocked)
    console.log(`🔎 Checking ${COMMON_SUBDOMAINS.length} common subdomains...`);
    const subdomainPromises = COMMON_SUBDOMAINS.map(async (subdomain) => {
      const subdomainHost = `${subdomain}.${domain}`;
      try {
        // Use GET with short timeout - more reliable than HEAD
        await axios.get(`https://${subdomainHost}`, {
          timeout: 4000,
          headers: { 'User-Agent': 'Mozilla/5.0 (compatible; CitationRocket/1.0)' },
          maxContentLength: 100000  // Limit response size
        });
        // Subdomain exists - try sitemap.xml first, then robots.txt
        let subSitemapUrls = await fetchSitemapUrls(subdomainHost);
        if (subSitemapUrls.length === 0) {
          // Try robots.txt for sitemap references
          subSitemapUrls = await fetchSitemapsFromRobots(subdomainHost);
        }
        if (subSitemapUrls.length > 0) {
          console.log(`   ✅ ${subdomainHost}: ${subSitemapUrls.length} URLs`);
        } else {
          console.log(`   📄 ${subdomainHost}: exists (will scrape pages)`);
        }
        // ALWAYS return subdomain as "exists" so we can scrape its pages
        return { subdomain: subdomainHost, urls: subSitemapUrls, exists: true };
      } catch {
        return { subdomain: subdomainHost, urls: [], exists: false };
      }
    });
    
    const subdomainResults = await Promise.all(subdomainPromises);
    for (const result of subdomainResults) {
      // Add sitemap URLs if found
      if (result.urls.length > 0) {
        allSitemapUrls = [...allSitemapUrls, ...result.urls];
      }
      // Include subdomain in discoveredSubdomains if it EXISTS (regardless of sitemap)
      if (result.exists) {
        discoveredSubdomains.push(result.subdomain);
      }
    }
    
    console.log(`📊 Total: ${allSitemapUrls.length} URLs across ${1 + discoveredSubdomains.length} domains`);
    if (discoveredSubdomains.length > 0) {
      console.log(`   Subdomains with content: ${discoveredSubdomains.join(', ')}`);
    }
    
    // Try to fetch additional pages for more signal data (main domain)
    let combinedHtml = html;
    let combinedText = text;
    const additionalPages = ['/blog', '/about', '/docs', '/resources', '/case-studies', '/faq', '/help'];
    
    for (const page of additionalPages) {
      try {
        const pageUrl = new URL(page, normalizedUrl).toString();
        const pageResponse = await axios.get(pageUrl, {
          timeout: 5000,
          headers: { 'User-Agent': 'Mozilla/5.0 (compatible; CitationRocket/1.0)' }
        });
        
        const pageHtml = pageResponse.data || '';
        const pageText = pageHtml
          .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '')
          .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '')
          .replace(/<[^>]+>/g, ' ')
          .replace(/\s+/g, ' ')
          .trim();
        
        combinedHtml += ' ' + pageHtml;
        combinedText += ' ' + pageText;
        console.log(`   📄 Scraped ${page}: ${pageText.length} chars`);
      } catch {
        // Page not accessible, skip silently
      }
    }
    
    // SUBDOMAIN PAGE SCRAPING: Fetch key pages from discovered subdomains
    for (const subdomain of discoveredSubdomains) {
      const subdomainPaths = ['/', '/tutorials', '/guides', '/courses', '/getting-started', '/documentation', '/how-to', '/training', '/learn'];
      for (const path of subdomainPaths) {
        try {
          const subUrl = `https://${subdomain}${path}`;
          const subResponse = await axios.get(subUrl, {
            timeout: 8000,
            maxRedirects: 5,
            headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36' }
          });
          const subText = (subResponse.data || '')
            .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '')
            .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '')
            .replace(/<[^>]+>/g, ' ')
            .replace(/\s+/g, ' ')
            .trim();
          combinedText += ' ' + subText;
          console.log(`   📄 Scraped ${subdomain}${path}: ${subText.length} chars`);
        } catch {
          // Skip inaccessible pages
        }
      }
    }
    
    // Sample some sitemap URLs (prioritize subdomain URLs, max 10 total)
    const prioritizedUrls = allSitemapUrls.filter(url => 
      discoveredSubdomains.some(sub => url.includes(sub))
    ).slice(0, 7);
    const mainDomainSample = mainSitemapUrls.slice(0, 3);
    const sampleUrls = [...prioritizedUrls, ...mainDomainSample].slice(0, 10);
    
    for (const sampleUrl of sampleUrls) {
      try {
        const sampleResponse = await axios.get(sampleUrl, {
          timeout: 3000,

        });

          .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '')
          .replace(/<[^>]+>/g, ' ')
          .replace(/\s+/g, ' ')
          .trim();
        combinedText += ' ' + sampleText;
      } catch {
        // Skip inaccessible URLs
      }
    }
    
    console.log(`✅ Scraped ${combinedHtml.length} bytes of HTML, ${combinedText.length} characters of text`);
    return { html: combinedHtml, text: combinedText, sitemapUrls: allSitemapUrls };
    
  } catch (error: any) {
    console.log(`⚠️ Website scraping failed: ${error.message}`);
    return { html: '', text: '', sitemapUrls: [] };
  }
}

// Anti-hallucination: Sanitize search volumes to ensure they use ranges
function sanitizeSearchVolume(volume: string): string {
  if (!volume) return 'Estimated: 500-2,000/month';
  
  // Already in correct format
  if (volume.toLowerCase().startsWith('estimated:') && volume.includes('-')) {
    return volume;
  }
  
  // Extract exact number and convert to range
  const numMatch = volume.match(/(\d[\d,]*)/);
  if (numMatch) {
    const num = parseInt(numMatch[1].replace(/,/g, ''));
    const lower = Math.floor(num * 0.7);
    const upper = Math.ceil(num * 1.3);
    return `Estimated: ${lower.toLocaleString()}-${upper.toLocaleString()}/month`;
  }
  
  return 'Estimated: 500-2,000/month';
}

// Strip Perplexity citation markers like [1], [2], [3] from text
function stripCitationMarkers(text: string): string {
  if (!text) return text;
  return text
    .replace(/\[\d+\]/g, '')  // Remove [1], [2], etc.
    .replace(/\s{2,}/g, ' ')  // Clean up double spaces
    .trim();
}

// Sanitize an array of strings - strip citation markers
function sanitizeStringArray(arr: string[]): string[] {
  if (!Array.isArray(arr)) return [];
  return arr.map(s => stripCitationMarkers(s || ''));
}

// Robust JSON parser with auto-repair for malformed AI responses
function parseRobustJSON(jsonString: string): any {
  // First try normal parse
  try {
    return JSON.parse(jsonString);
  } catch (parseError) {
    console.log('⚠️ JSON parse failed, attempting repair...');
    
    let repaired = jsonString.trim();
    
    // STEP 1: Fix truncated strings by checking for unclosed quotes
    let inString = false;
    let lastQuotePos = -1;
    for (let i = 0; i < repaired.length; i++) {
      if (repaired[i] === '"' && (i === 0 || repaired[i - 1] !== '\\')) {
        inString = !inString;
        if (inString) lastQuotePos = i;
      }
    }
    if (inString) {
      // We're in an unclosed string - find a safe truncation point
      // Truncate to last complete element before this string
      const beforeString = repaired.substring(0, lastQuotePos);
      const lastSafePoint = Math.max(
        beforeString.lastIndexOf('",'),
        beforeString.lastIndexOf('"],'),
        beforeString.lastIndexOf('"}'),
        beforeString.lastIndexOf('"]')
      );
      if (lastSafePoint > 0) {
        repaired = repaired.substring(0, lastSafePoint + 2);
      } else {
        repaired += '"';
      }
    }
    
    // STEP 2: Handle truncated arrays - find last complete element
    // Look for patterns like ["item1", "item2", "incomple...
    const arrayPattern = /\[\s*"[^"]*"(?:\s*,\s*"[^"]*")*\s*,?\s*$/;
    if (arrayPattern.test(repaired)) {
      // Find last complete string in array
      const matches = repaired.match(/"[^"]*"/g);
      if (matches && matches.length > 0) {
        const lastCompleteString = matches[matches.length - 1];
        const lastCompletePos = repaired.lastIndexOf(lastCompleteString);
        if (lastCompletePos > 0) {
          repaired = repaired.substring(0, lastCompletePos + lastCompleteString.length);
        }
      }
    }
    
    // STEP 3: For objects with arrays, find last complete array element
    if (repaired.includes('[')) {
      // Check each array for incomplete last element
      let result = repaired;
      const arrayStart = result.indexOf('[');
      if (arrayStart > 0) {
        // Find the last complete "string", pattern in any array
        const lastGoodComma = result.lastIndexOf('",');
        const lastGoodBracket = result.lastIndexOf('"]');
        const lastGood = Math.max(lastGoodComma, lastGoodBracket);
        
        if (lastGood > arrayStart) {
          // Check if there's incomplete content after
          const afterGood = result.substring(lastGood + 2).trim();
          if (afterGood && !afterGood.startsWith(']') && !afterGood.startsWith('}') && !afterGood.startsWith('"')) {
            result = result.substring(0, lastGood + 2);
          }
        }
      }
      repaired = result;
    }
    
    // STEP 4: Remove incomplete objects in arrays
    if (repaired.includes('[{')) {
      const lastCompleteObject = repaired.lastIndexOf('},');
      const lastObjectClose = repaired.lastIndexOf('}]');
      if (lastCompleteObject > lastObjectClose && lastCompleteObject > 0) {
        repaired = repaired.substring(0, lastCompleteObject + 1);
      }
    }
    
    // STEP 5: For plain objects, remove trailing incomplete properties
    if (repaired.startsWith('{') && !repaired.includes('[')) {
      const lastQuote = repaired.lastIndexOf('"');
      const lastColon = repaired.lastIndexOf(':');
      const lastComma = repaired.lastIndexOf(',');
      
      if (lastColon > lastComma && lastQuote > lastColon) {
        repaired = repaired.substring(0, lastComma > 0 ? lastComma : repaired.lastIndexOf('}'));
      }
    }
    
    // STEP 6: Remove trailing commas before closing brackets
    repaired = repaired.replace(/,(\s*)$/, '$1');
    repaired = repaired.replace(/,(\s*[}\]])/g, '$1');
    
    // STEP 7: Balance brackets and braces
    const openBrackets = (repaired.match(/\[/g) || []).length;
    const closeBrackets = (repaired.match(/\]/g) || []).length;
    const openBraces = (repaired.match(/\{/g) || []).length;
    const closeBraces = (repaired.match(/\}/g) || []).length;
    
    repaired += ']'.repeat(Math.max(0, openBrackets - closeBrackets));
    repaired += '}'.repeat(Math.max(0, openBraces - closeBraces));
    
    try {
      const result = JSON.parse(repaired);
      console.log('⚠️ JSON repaired (partial data may be missing)');
      return result;
    } catch (repairError) {
      console.error('❌ JSON repair failed:', repairError);
      throw parseError; // Throw original error
    }
  }
}

// Validate competitor analysis has required fields with actual data
function validateCompetitorAnalysis(data: any): boolean {
  if (!data || typeof data !== 'object') return false;
  
  // Check required arrays exist and have at least some data
  const requiredArrays = ['mainCompetitors', 'competitorStrengths', 'competitorWeaknesses', 'marketOpportunities'];
  for (const field of requiredArrays) {
    if (!Array.isArray(data[field])) return false;
    // At least one field should have data for valid analysis
  }
  
  // Valid if at least one array has data
  const totalItems = requiredArrays.reduce((sum, field) => sum + (data[field]?.length || 0), 0);
  return totalItems > 0;
}

// Sanitize topics to remove hallucinated exact metrics and citation markers
function sanitizeTopics(topics: any[]): any[] {
  const validContentTypes = ['tutorial', 'case-study', 'faq', 'data-driven', 'framework', 'how-to'];
  const validDifficulties = ['Easy', 'Medium', 'Hard'];
  const validEfforts = ['Low', 'Medium', 'High'];
  const validPriorityTiers = ['Quick Win', 'High Impact', 'Long-term'];
  
  return topics.map(topic => ({
    ...topic,
    topic: stripCitationMarkers(topic.topic || ''),
    searchVolume: sanitizeSearchVolume(topic.searchVolume || ''),
    justification: stripCitationMarkers((topic.justification || '').replace(/cited?\s*\d+\s*times?/gi, 'has citation potential')),
    competitorGap: stripCitationMarkers(topic.competitorGap || ''),
    // Actionable metrics (replaces meaningless scoreImpact %)
    difficulty: validDifficulties.includes(topic.difficulty) ? topic.difficulty : 'Medium',
    effort: validEfforts.includes(topic.effort) ? topic.effort : 'Medium',
    priorityTier: validPriorityTiers.includes(topic.priorityTier) ? topic.priorityTier : 'High Impact',
    timeEstimate: topic.timeEstimate || '3-5 hours',
    // Branding and platform targeting
    isBranded: topic.isBranded === true,
    targetPlatform: ['chatgpt', 'perplexity', 'gemini', 'claude'].includes(topic.targetPlatform) ? topic.targetPlatform : 'chatgpt',
    gapFilled: topic.gapFilled || 'tutorials',
    // Content type and platforms
    contentType: validContentTypes.includes(topic.contentType) ? topic.contentType : 'how-to',
    platforms: Array.isArray(topic.platforms) ? topic.platforms : [topic.targetPlatform || 'chatgpt']
  }));
}
